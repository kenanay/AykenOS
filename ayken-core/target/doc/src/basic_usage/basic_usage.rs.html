1//! Basic ABDF and BCIB usage examples
2
3use abdf_builder::{AbdfBuilder, decode_abdf};
4use abdf::segment::{SegmentKind, MetaContainer};
5use bcib::{BcibHeader, BcibInstruction, BcibOpcode};

7fn main() -> Result<(), Box<dyn std::error::Error>> {
8    println!("=== ABDF Basic Usage ===");
9    abdf_example()?;
10    
11    println!("\n=== BCIB Basic Usage ===");
12    bcib_example()?;
13    
14    Ok(())
15}

17fn abdf_example() -> Result<(), Box<dyn std::error::Error>> {
18    // Create a new ABDF builder
19    let mut builder = AbdfBuilder::new();
20    
21    // Add some strings to the string pool
22    let users_name = builder.intern_string("users");
23    let table_type = builder.intern_string("table/generic");
24    let schema_str = builder.intern_string("id:u64,name:string,email:string");
25    
26    // Create metadata for a user table
27    let user_meta = MetaContainer {
28        name_idx: users_name,
29        type_idx: table_type,
30        schema_idx: schema_str,
31        permissions: 0,
32        embedding_idx: 0,
33    };
34    
35    // Sample user data (in a real scenario, this would be properly serialized)
36    let user_data = b"1,John Doe,john@example.com\n2,Jane Smith,jane@example.com\n";
37    
38    // Add the user table segment
39    let segment_idx = builder.add_segment(SegmentKind::Tabular(user_meta), user_data);
40    println!("Added user table segment at index: {}", segment_idx);
41    
42    // Add a raw data segment
43    let raw_data = &[0xDE, 0xAD, 0xBE, 0xEF, 0xCA, 0xFE, 0xBA, 0xBE];
44    let raw_idx = builder.add_segment(SegmentKind::Raw, raw_data);
45    println!("Added raw data segment at index: {}", raw_idx);
46    
47    // Build the ABDF buffer
48    let buffer = builder.build();
49    println!("Built ABDF buffer with {} bytes", buffer.len());
50    
51    // Decode the buffer
52    let view = decode_abdf(&buffer)?;
53    println!("Decoded ABDF buffer successfully");
54    println!("Header version: {}", view.header.version);
55    println!("Number of segments: {}", view.segments.len());
56    
57    // Access the user table
58    if let Some(name) = view.segment_name(0) {
59        println!("Segment 0 name: {}", name);
60    }
61    
62    if let Some(data) = view.segment_data(0) {
63        println!("Segment 0 data: {}", String::from_utf8_lossy(data));
64    }
65    
66    // Access the raw data
67    if let Some(data) = view.segment_data(1) {
68        println!("Segment 1 data: {:02X?}", data);
69    }
70    
71    Ok(())
72}

74fn bcib_example() -> Result<(), Box<dyn std::error::Error>> {
75    // Create BCIB header
76    let mut header = BcibHeader::new();
77    header.instruction_count = 4;
78    header.string_pool_offset = 16 + (4 * 8); // header + 4 instructions
79    
80    println!("BCIB Header:");
81    println!("  Magic: {:?}", std::str::from_utf8(&header.magic).unwrap_or("Invalid"));
82    println!("  Version: {}", header.version);
83    println!("  Instruction count: {}", header.instruction_count);
84    
85    // Create instructions
86    let instructions = vec![
87        // Select container 0 (users table)
88        BcibInstruction::new(BcibOpcode::CtxSelect, 0, 0, 0, 0),
89        
90        // Query data using string at index 0
91        BcibInstruction::new(BcibOpcode::DataQuery, 0, 0, 0, 0),
92        
93        // Render UI scene
94        BcibInstruction::new(BcibOpcode::UiRender, 0, 1, 0, 0),
95        
96        // End execution
97        BcibInstruction::new(BcibOpcode::End, 0, 0, 0, 0),
98    ];
99    
100    println!("\nInstructions:");
101    for (i, instr) in instructions.iter().enumerate() {
102        println!("  {}: {:?} (args: {}, {}, {})", 
103                 i, instr.opcode, instr.arg0, instr.arg1, instr.arg2);
104    }
105    
106    // String pool
107    let string_pool = "SELECT * FROM users WHERE active = 1\0";
108    println!("\nString pool: {:?}", string_pool);
109    
110    // In a real implementation, you would serialize this to a binary buffer
111    println!("\nTotal BCIB size would be: {} bytes", 
112             std::mem::size_of::<BcibHeader>() + 
113             instructions.len() * std::mem::size_of::<BcibInstruction>() + 
114             string_pool.len());
115    
116    Ok(())
117}
