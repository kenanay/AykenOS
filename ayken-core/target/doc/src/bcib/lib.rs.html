1//! BCIB (Binary CLI Instruction Buffer) Core Format (v0.2)
2//!
3//! Bu crate, AykenOS'un CLI komutlarını temsil eden binary formatı tanımlar.
4//! Faz 2 hedeflerine uygun olarak, bu format üst düzey, veri odaklı
5//! komutları içerir ve bir başlık (header) ile versiyonlanır.
6//!
7//! # Örnek Kullanım
8//!
9//! ```
10//! use bcib::{BcibHeader, BcibInstruction, BcibOpcode};
11//!
12//! // Header oluştur
13//! let mut header = BcibHeader::new();
14//! assert!(header.is_valid());
15//! assert_eq!(header.version, 2);
16//!
17//! // Instruction oluştur
18//! let instr = BcibInstruction::new(BcibOpcode::DataQuery, 0, 123, 0, 0);
19//! assert_eq!(instr.opcode, BcibOpcode::DataQuery);
20//! assert_eq!(instr.arg0, 123);
21//! ```
22
23pub const BCIB_MAGIC: &[u8; 4] = b"BCIB";
24pub const BCIB_VERSION: u16 = 2; // 0.2
25
26/// BCIB buffer'ının başlık yapısı.
27#[repr(C)]
28#[derive(Debug, Clone, Copy, PartialEq, Eq)]
29pub struct BcibHeader {
30    /// Magic number, her zaman "BCIB" olmalı.
31    pub magic: [u8; 4],
32    /// BCIB format versiyonu (örn: 2 -> 0.2).
33    pub version: u16,
34    /// Header'dan sonraki instruction sayısını belirtir.
35    pub instruction_count: u16,
36    /// Argümanlar için kullanılan string pool'un başlangıç ofseti (header'dan itibaren).
37    /// Eğer 0 ise, string pool yok demektir.
38    pub string_pool_offset: u32,
39    /// Gelecekteki kullanım için ayrılmıştır.
40    pub reserved: u32,
41}
42
43impl BcibHeader {
44    pub fn new() -> Self {
45        Self {
46            magic: *BCIB_MAGIC,
47            version: BCIB_VERSION,
48            instruction_count: 0,
49            string_pool_offset: 0,
50            reserved: 0,
51        }
52    }
53
54    pub fn is_valid(&self) -> bool {
55        self.magic == *BCIB_MAGIC
56    }
57}

59/// Faz 2 DSL'ine uygun BCIB Opcode tanımları.
60#[repr(u8)]
61#[derive(Debug, Clone, Copy, PartialEq, Eq)]
62pub enum BcibOpcode {
63    /// No operation: Hiçbir işlem yapmaz.
64    Nop = 0x00,
65
66    // --- Context Management ---
67    /// Aktif veri konteynerini seçer. Arg0: ABDF segment index.
68    CtxSelect = 0x01,
69
70    // --- Data Operations ---
71    /// Yeni bir veri konteyneri oluşturur. Arg0: ABDF meta_idx.
72    DataCreate = 0x10,
73    /// Seçili konteynere veri ekler. Arg0: eklenecek veri için ABDF segment_idx.
74    DataAdd = 0x11,
75    /// Seçili konteynerde sorgu çalıştırır. Arg0: string pool'daki sorgu metni index'i.
76    DataQuery = 0x12,
77    /// Seçili konteynerdeki veriyi günceller. (Gelecek için placeholder)
78    DataUpdate = 0x13,
79    /// Seçili konteynerden veri siler. (Gelecek için placeholder)
80    DataDelete = 0x14,
81
82    // --- UI Operations ---
83    /// Bir UI sahnesini render eder. Arg0: ABDF'deki UiScene meta_idx'i.
84    UiRender = 0x20,
85    /// Bir UI bileşenine olay gönderir. (Gelecek için placeholder)
86    UiEvent = 0x21,
87
88    // --- AI Operations ---
89    /// Bir AI modülüne soru sorar. Arg0: string pool'daki soru metni index'i.
90    AiAsk = 0x30,
91
92    // --- System Operations ---
93    /// Sistem bilgisi sorgular. (Gelecek için placeholder)
94    SysInfo = 0x40,
95    /// Programı veya script'i sonlandırır.
96    End = 0xFF,
97}

99/// Tek bir BCIB instruction'ı (8 byte).
100/// Argümanlar genellikle ABDF veya BCIB string pool'daki index'lerdir.
101#[repr(C)]
102#[derive(Debug, Clone, Copy, PartialEq, Eq)]
103pub struct BcibInstruction {
104    pub opcode: BcibOpcode,
105    /// Opcode'a özel bayraklar.
106    pub flags: u8,
107    /// Argüman 1.
108    pub arg0: u16,
109    /// Argüman 2.
110    pub arg1: u16,
111    /// Argüman 3.
112    pub arg2: u16,
113}
114
115impl BcibInstruction {
116    pub fn new(opcode: BcibOpcode, flags: u8, arg0: u16, arg1: u16, arg2: u16) -> Self {
117        Self { opcode, flags, arg0, arg1, arg2 }
118    }
119
120    pub fn nop() -> Self {
121        Self::new(BcibOpcode::Nop, 0, 0, 0, 0)
122    }
123}

145    fn header_validity() {
146        let mut header = BcibHeader::new();
147        assert!(header.is_valid());
148

151    }
