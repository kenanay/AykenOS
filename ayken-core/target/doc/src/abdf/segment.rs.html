1//! ABDF Segment Model (v0.2)
2//!
3//! Bu modül, ABDF buffer'ı içindeki "segment" ve "container"
4//! kavramlarını tanımlar. Meta-veri ve fiziksel veri ayrılmıştır.
5//!
6//! # Örnek Kullanım
7//!
8//! ```
9//! use abdf::segment::{SegmentDescriptor, SegmentKind, MetaContainer};
10//!
11//! // Meta container oluştur
12//! let meta = MetaContainer {
13//!     name_idx: 0,
14//!     type_idx: 1,
15//!     schema_idx: 2,
16//!     permissions: 0,
17//!     embedding_idx: 0,
18//! };
19//!
20//! // Segment kind oluştur
21//! let kind = SegmentKind::Tabular(meta);
22//! assert!(kind.is_tabular());
23//!
24//! // Segment descriptor oluştur
25//! let desc = SegmentDescriptor::new(0, 1024, 4096);
26//! assert_eq!(desc.offset, 1024);
27//! assert_eq!(desc.length, 4096);
28//! ```

30/// Bir veri konteynerinin meta-verilerini tutan genel yapı.
31/// İndeksler (idx), string pool'daki karşılıklarına işaret eder.
32#[derive(Debug, Clone, PartialEq, Eq)]
33pub struct MetaContainer {
34    /// Konteynerin adı (örn: "users", "system_log")
35    pub name_idx: u32,
36    /// Konteynerin tipi (örn: "table/sql", "log/syslog")
37    pub type_idx: u32,
38    /// Veri şeması (örn: "id:int,name:string" veya JSON şeması)
39    pub schema_idx: u32,
40    /// İzinler (ileride kullanılacak - placeholder)
41    pub permissions: u64,
42    /// Anlamsal embedding verisi indeksi (ileride kullanılacak)
43    pub embedding_idx: u32,
44}
45
46/// ABDF'de hangi tür segment olduğunu belirten enum (v0.2).
47/// Bu yapı, segmentin anlamsal türünü ve ilişkili meta-verisini içerir.
48/// Bu enum'lar bir "Meta Tablosu"nda saklanır.
49#[derive(Debug, Clone, PartialEq, Eq)]
50pub enum SegmentKind {
51    /// Zengin meta-veriye sahip konteyner bazlı segmentler.
52    Tabular(MetaContainer),
53    Log(MetaContainer),
54    Text(MetaContainer),
55    UiScene(MetaContainer),
56    GpuBuffer(MetaContainer),
57
58    /// Basit, meta-verisiz ham byte verisi.
59    Raw,
60}
61
62/// ABDF segment descriptor (v0.2).
63///
64/// Bu yapı, bir segment verisinin buffer içindeki fiziksel konumunu
65/// ve meta-verisine olan bağlantısını belirtir.
66#[repr(C)]
67#[derive(Debug, Clone, Copy, PartialEq, Eq)]
68pub struct SegmentDescriptor {
69    /// Meta tablosundaki `SegmentKind`'e işaret eden indeks.
70    pub meta_idx: u32,
71
72    /// Segment verisinin buffer içindeki byte offset'i.
73    pub offset: u64,
74
75    /// Segment verisinin byte cinsinden uzunluğu.
76    pub length: u64,
77}
78
79impl SegmentDescriptor {
80    /// Yeni bir segment descriptor oluşturur.
81    pub fn new(meta_idx: u32, offset: u64, length: u64) -> Self {
82        Self {
83            meta_idx,
84            offset,
85            length,
86        }
87    }
88}
89
90impl SegmentKind {
91    /// Bu segmentin tabular veri taşıyıp taşımadığını söyler.
92    pub fn is_tabular(&self) -> bool {
93        matches!(self, Self::Tabular(_))
94    }
95
96    /// Bu segment bir UI sahnesi mi?
97    pub fn is_ui_scene(&self) -> bool {
98        matches!(self, Self::UiScene(_))
99    }
100}
